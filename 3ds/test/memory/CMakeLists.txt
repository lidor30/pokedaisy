cmake_minimum_required(VERSION 3.18)
project(pd_rom_memory_test C CXX ASM)
enable_testing()
include("${CMAKE_CURRENT_LIST_DIR}/../../emu/gpsp.cmake")
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
add_executable(rom_memory_test rom_memory_test.c)
target_link_libraries(rom_memory_test PRIVATE pd_gpsp Threads::Threads ZLIB::ZLIB m)
target_compile_options(rom_memory_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_definitions(rom_memory_test PRIVATE PD_TEST_THREAD_FAILURE)
    target_link_options(rom_memory_test PRIVATE -Wl,--wrap=pthread_create)
endif()
add_executable(refresh_test refresh_test.c)
target_compile_options(refresh_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_executable(pixels_test pixels_test.c)
target_compile_options(pixels_test PRIVATE -O2 -Wall -Wextra -Werror -UNDEBUG)
add_executable(thread_start_test thread_start_test.c "${GPSP_DIR}/video_thread_sys.c")
target_include_directories(thread_start_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/thread_stubs")
target_compile_definitions(thread_start_test PRIVATE _3DS)
target_compile_options(thread_start_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_executable(log_buffer_test log_buffer_test.c)
target_compile_options(log_buffer_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME companion_refresh COMMAND refresh_test)
add_test(NAME texture_pixels COMMAND pixels_test)
add_test(NAME render_thread_start COMMAND thread_start_test)
add_test(NAME deferred_log_buffer COMMAND log_buffer_test)
