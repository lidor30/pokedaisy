# The lock and co-op's retail check side by side: either order installs (co-op's CRC check of ScriptContext_SetupScript
# reads the file's bytes under the lock's own head patch), the lock's block is in (0x08FFE000), the ROM still reads as the
# file, each uninstalls alone and the other stays, and the hook holds with co-op installed.
# want: lock: installed
# want: coop: installed
# want: lock: held 6 kind 1
# want: lock: uninstalled, rom patched: no
# want: coop: uninstalled, rom patched: no
# want: romcheck raw bus: 0 byte(s) differ
# want: romcheck overlay: 0 byte(s) differ
# want: romcheck swapped: 0 byte(s) differ
lockinstall
coopinstall
lockactive 1
call 0x08098EF8 0x081FC7C2
lockheld
expect32 0x08098EFC 0x08FFE001
romcheck $EMERALD_ROM
lockuninstall
expect8 0x08FFE000 0xFF 16
expect32 0x08098EF8 0x1C05B530
coopuninstall
coopinstall
lockinstall
lockactive 1
call 0x08098EF8 0x081FC7C2
lockheld
coopuninstall
expect32 0x08098EFC 0x08FFE001
lockuninstall
expect32 0x08098EF8 0x1C05B530
romcheck $EMERALD_ROM
